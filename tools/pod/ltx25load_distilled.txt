# Rebuilt 2026-09-08 from the verified base-model manifest; not a byte-for-byte
# recovery of the original file. The separate MSR downloader remains unchanged.
# Paste/run this entire file on the pod; Python 3.10+ with venv is required.
# Optional environment: COMFY_ROOT=/workspace/ComfyUI, HF_TOKEN (read token),
# HF_HOME=/workspace/hf_cache, LTX_PYTHON=python3.
# Accept any required Hugging Face repository terms before downloading.
#
# Five pinned BF16 artifacts (~71.11 GB / 66.23 GiB total):
# distilled transformer, Gemma4 12B encoder, video VAE, audio VAE,
# and native spatial x2 latent upscaler (for two-stage workflows).
# No MSR LoRA or MSR/PromptRelay custom nodes are downloaded or installed.
# Valid existing models are reused after full SHA-256 verification.
# Existing extra models/custom nodes are left in place.
# Allow another 2 GiB free per destination filesystem for staging overhead.
# ComfyUI must already be installed with native LTX-2.5 support.
# This model setup does not modify workflows, restart ComfyUI or render a video.
# Source: https://huggingface.co/Lightricks/LTX-2.5
# The temporary downloader environment is version/hash locked and isolated
# from ComfyUI's Python. Keep the child-shell wrapper so failure does not close SSH.

def sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for block in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()

def is_complete(path: Path, expected_size: int, expected_sha256: str) -> bool:
    # A parseable header and byte count cannot detect corrupt tensor data.
    # Hash the entire file against the pinned official artifact; no untrusted
    # header length is used to allocate memory.
    if not path.is_file() or path.stat().st_size != expected_size:
        return False
    return sha256(path) == expected_sha256

def ensure_model(repo: str, remote: str, directory: Path, expected_size: int,
                 expected_sha256: str, revision: str) -> Path:
    directory.mkdir(parents=True, exist_ok=True)
    destination = directory / Path(remote).name
    if is_complete(destination, expected_size, expected_sha256):
        print(f"OK (SHA-256 verified): {destination.name}")
        return destination

    # Stage on the destination filesystem. Leave any existing model untouched
    # until the complete replacement passes its pinned checksum.
    if destination.exists():
        print(f"Does not match pinned artifact; retaining until replacement validates: {destination.name}")
    staging = directory / ".ltx25-downloads" / revision

    print(f"Downloading: {destination.name}")
    downloaded = Path(
        hf_hub_download(
            repo_id=repo,
            filename=remote,
            revision=revision,
            local_dir=str(staging),
            token=token,
        )
    )
    if not downloaded.resolve().is_relative_to(staging.resolve()):
        raise RuntimeError(f"Unexpected download location; original model retained: {destination.name}")
    if not is_complete(downloaded, expected_size, expected_sha256):
        # Remove only the corrupt staged download, so a later retry cannot
        # reuse it via the Hub's local-dir metadata. Never unlink destination.
        downloaded.unlink()
        raise RuntimeError(f"SHA-256/size validation failed; original model retained: {destination.name}")
    downloaded.replace(destination)
    return destination

def check_environment() -> None:
    root = models.parent
    if not root.is_absolute() or not root.is_dir() or not (root / "main.py").is_file():
        raise RuntimeError("COMFY_ROOT must be the absolute path of an existing ComfyUI installation containing main.py.")
    print(f"Model destination: {models}")


def pending_models() -> list:
    return [entry for entry in FILES if not is_complete(entry[2] / Path(entry[1]).name, entry[3], entry[4])]


def check_disk_space(pending: list) -> None:
    # Group by filesystem, allow full staged replacements while originals remain,
    # and reserve 2 GiB per filesystem. Do not count partial downloads as free space.
    budgets = {}
    for _, remote, directory, size, _, _ in pending:
        directory.mkdir(parents=True, exist_ok=True)
        with tempfile.TemporaryFile(dir=directory):
            pass
        device = directory.stat().st_dev
        path, needed = budgets.get(device, (directory, 2 * 1024**3))
        budgets[device] = (path, needed + size)
    for path, needed in budgets.values():
        free = shutil.disk_usage(path).free
        if free < needed:
            raise RuntimeError(f"Insufficient staging space on {path}: need {needed / 1024**3:.1f} GiB free, have {free / 1024**3:.1f} GiB.")


def check_remote_access(pending: list) -> None:
    # HEAD all required replacements before starting the first large transfer.
    for repo, remote, _, size, digest, revision in pending:
        try:
            metadata = get_hf_file_metadata(hf_hub_url(repo, remote, revision=revision), token=token)
        except Exception:
            raise RuntimeError(f"Cannot access pinned {repo}/{remote}. Check HF_TOKEN and repository access; no new model transfer started.") from None
        if metadata.size != size or metadata.etag != digest or metadata.commit_hash != revision:
            raise RuntimeError(f"Remote metadata disagrees with the pinned manifest: {remote}. Review the loader before downloading.")


def download_models() -> None:
    # Keep every gate outside ensure_model's transfer loop.
    check_environment()
    pending = pending_models()
    check_disk_space(pending)
    check_remote_access(pending)
    for entry in pending:
        ensure_model(*entry)
    for _, remote, directory, size, digest, _ in FILES:
        if not is_complete(directory / Path(remote).name, size, digest):
            raise RuntimeError(f"Final model validation failed: {remote}")
    print("PASS: all five LTX-2.5 distilled model files match their pinned sizes and SHA-256 digests.")
    print("Next: select these models in a native LTX-2.5 distilled workflow and run a test clip.")
    print("No app integration, custom-node installation, or GPU render was performed.")


if __name__ == "__main__":
    try:
        if "--preflight" in sys.argv:
            check_environment()
        else:
            from huggingface_hub import hf_hub_download, get_hf_file_metadata, hf_hub_url
            token = os.environ.get("HF_TOKEN") or None
            download_models()
    except Exception as error:
        print(f"FAILED: {error}", file=sys.stderr)
        raise SystemExit(1)
